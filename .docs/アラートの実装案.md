resources:
  alerts:
    table_freshness_alert:
      display_name: "[${bundle.target}] テーブル鮮度チェック"
      warehouse_id: ${var.warehouse_id}
      query_text: |
        SELECT
          table_name,
          max_loaded_at,
          timestampdiff(HOUR, max_loaded_at, current_timestamp()) AS lag_hours
        FROM ops.monitoring.table_freshness
        WHERE timestampdiff(HOUR, max_loaded_at, current_timestamp()) > 24
      evaluation:
        source:
          name: lag_hours
          aggregation: COUNT
        comparison_operator: GREATER_THAN
        threshold:
          value:
            double_value: 0
        empty_result_state: OK
        notification:
          notify_on_ok: true
          retrigger_seconds: 3600
          subscriptions:
            - user_email: someone@example.com
      schedule:
        quartz_cron_schedule: "0 0 8 * * ?"
        timezone_id: Asia/Tokyo
      custom_summary: "[{{ALERT_STATUS}}] {{ALERT_NAME}}"
      custom_description: |
        # ↓ 以下のパターンのいずれかを入れる
