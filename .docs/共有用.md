KEYS = [
    "spark.sql.session.timeZone",
    "spark.sql.ansi.enabled",
    "spark.sql.legacy.timeParserPolicy",
    "spark.sql.datetime.java8API.enabled",
    "spark.sql.execution.arrow.pyspark.enabled",
    "spark.sql.storeAssignmentPolicy",
    "spark.sql.legacy.parquet.datetimeRebaseModeInRead",
    "spark.sql.legacy.parquet.datetimeRebaseModeInWrite",
    "spark.sql.parquet.int96RebaseModeInRead",
]
for k in KEYS:
    try:
        print(f'.config("{k}", "{spark.conf.get(k)}")')
    except Exception:
        print(f"# {k}: unset")

import java.util  # noqa  -- 代わりに↓
print("JVM TZ:", spark.sparkContext._jvm.java.util.TimeZone.getDefault().getID())
print("Spark:", spark.version)
